# Nginx configuration for danielmangur.com/screener
# Option A: With rewrite (strips /screener prefix before passing to Express)
location /screener {
    rewrite ^/screener/?(.*)$ /$1 break;
    proxy_pass http://127.0.0.1:3045;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection 'upgrade';
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_cache_bypass $http_upgrade;
    client_max_body_size 10M;
}

# Note: The Express app natively handles both root (/) and (/screener)
# so both Option A (with rewrite) and standard proxy_pass work seamlessly.
